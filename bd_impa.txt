-- ENUMs
CREATE TYPE especie_mascota AS ENUM ('perro', 'gato', 'conejo', 'hamster', 'otro');
CREATE TYPE tamano_mascota AS ENUM ('pequeño', 'mediano', 'grande');
CREATE TYPE sexo_mascota AS ENUM ('macho', 'hembra');
CREATE TYPE estado_adopcion AS ENUM ('disponible', 'en_proceso', 'adoptado');
CREATE TYPE estado_cita AS ENUM ('programada', 'completada', 'cancelada', 'no_asistio');
CREATE TYPE estado_documentacion AS ENUM ('pendiente', 'verificada', 'rechazada');
CREATE TYPE estado_solicitud_adopcion AS ENUM ('en_revision','aprobada','rechazada','cancelada');
CREATE TYPE decision_adopcion AS ENUM ('aceptada','rechazada');
CREATE TYPE tipo_notificacion AS ENUM ('seguimiento','aceptacion', 'rechazo', 'recordatorio_cita', 'creacion_cita', 'actualizacion_cita', 'documento_verificado', 'documento_rechazado', 'nueva_solicitud', 'expediente_creado');

CREATE TABLE roles (
    id_rol SERIAL PRIMARY KEY,
    nombre_rol VARCHAR(50) UNIQUE NOT NULL
);

-- USUARIOS
CREATE TABLE usuarios (
  id_usuario SERIAL PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  correo_electronico VARCHAR(100) UNIQUE NOT NULL,
  contrasena TEXT NOT NULL,
  fecha_nacimiento DATE NOT NULL,
  telefono VARCHAR(20),
  documentacion_verificada estado_documentacion DEFAULT 'pendiente',
  foto_perfil_base64 TEXT,
  id_rol INT REFERENCES roles(id_rol) ON DELETE CASCADE,
  activo BOOLEAN DEFAULT TRUE,
  fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  fecha_actualizacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);
COMMENT ON TABLE usuarios IS 'Tabla principal que contiene todos los usuarios del sistema';

CREATE TABLE direcciones (
    id_direccion SERIAL PRIMARY KEY,
    id_usuario INT REFERENCES usuarios(id_usuario) ON DELETE CASCADE, 
    calle VARCHAR(200) NOT NULL,
    colonia VARCHAR(100) NOT NULL,
    codigo_postal VARCHAR(10) NOT NULL,
    ciudad VARCHAR(100) NOT NULL,
    estado VARCHAR(100) NOT NULL,
    pais VARCHAR(100) NOT NULL DEFAULT 'México',
    tipo_direccion VARCHAR(50) DEFAULT 'domicilio', 
    es_principal BOOLEAN DEFAULT TRUE, 
    CONSTRAINT uq_direccion_principal UNIQUE (id_usuario, es_principal) 
    DEFERRABLE INITIALLY IMMEDIATE 
);
COMMENT ON TABLE direcciones IS 'Direcciones asociadas a los usuarios (domicilio, envío, etc.).';

-- EMPLEADOS
CREATE TABLE empleados (
  id_empleado SERIAL PRIMARY KEY,
  id_usuario INT REFERENCES usuarios(id_usuario) ON DELETE CASCADE,
  numero_empleado VARCHAR(20) UNIQUE NOT NULL,
  fecha_contratacion DATE DEFAULT CURRENT_DATE,
  cedula_profesional VARCHAR(20),
  licenciatura VARCHAR(100),
  especialidad VARCHAR(100),
  activo BOOLEAN DEFAULT TRUE,
  CONSTRAINT uq_empleados_cedula_profesional UNIQUE (cedula_profesional)
);
COMMENT ON TABLE empleados IS 'Empleados de IMPA; veterinarios incluidos (cédula, especialidad)';

-- =========================
--  Función y Trigger para numero_empleado automático
-- =========================
CREATE OR REPLACE FUNCTION generar_siguiente_numero_empleado()
RETURNS TRIGGER AS $$
DECLARE
    ultimo_numero INT;
    siguiente_numero_str VARCHAR;
BEGIN
    -- Obtener el número más alto de la columna numero_empleado
    SELECT COALESCE(MAX(CAST(SUBSTRING(numero_empleado FROM 4) AS INTEGER)), 0)
    INTO ultimo_numero
    FROM empleados;

    -- Generar el siguiente número en formato 'EMP' + 3 dígitos con padding de ceros
    siguiente_numero_str := 'EMP' || LPAD((ultimo_numero + 1)::TEXT, 3, '0');
    NEW.numero_empleado := siguiente_numero_str;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_set_numero_empleado
BEFORE INSERT ON empleados
FOR EACH ROW
EXECUTE FUNCTION generar_siguiente_numero_empleado();

-- MASCOTAS
CREATE TABLE mascotas (
  id_mascota SERIAL PRIMARY KEY,
  nombre VARCHAR(50) NOT NULL,
  especie especie_mascota NOT NULL,
  raza VARCHAR(50),
  edad_en_meses INTEGER NOT NULL CHECK (edad_en_meses >= 0),
  color VARCHAR(30) NOT NULL,
  tamano tamano_mascota NOT NULL,
  sexo sexo_mascota NOT NULL,
  descripcion TEXT,
  vacunado BOOLEAN DEFAULT FALSE,
  esterilizado BOOLEAN DEFAULT FALSE,
  estado_adopcion estado_adopcion DEFAULT 'disponible',
  fecha_ingreso DATE DEFAULT CURRENT_DATE,
  fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  fecha_actualizacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  creado_por INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL,
  activo BOOLEAN DEFAULT TRUE
);
CREATE INDEX idx_mascotas_estado_activo ON mascotas(estado_adopcion, activo);
CREATE INDEX idx_mascotas_especie_estado ON mascotas(especie, estado_adopcion);
COMMENT ON TABLE mascotas IS 'Información de todas las mascotas disponibles para adopción';

-- IMAGENES DE MASCOTAS
CREATE TABLE imagenes_mascotas (
  id_imagen_mascota SERIAL PRIMARY KEY,
  id_mascota INT REFERENCES mascotas(id_mascota) ON DELETE SET NULL,
  imagen_base64 TEXT
);
COMMENT ON TABLE mascotas IS 'Fotos de la mascota';

CREATE TABLE servicios (
  id_servicio SERIAL PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  descripcion TEXT,
  costo_base DECIMAL(10,2) DEFAULT 0.00,
  duracion_estimada_min INTEGER,
  requiere_mascota BOOLEAN DEFAULT TRUE,
  activo BOOLEAN DEFAULT TRUE,
  fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- CITAS
CREATE TABLE citas (
  id_cita SERIAL PRIMARY KEY,
  id_usuario INT NOT NULL REFERENCES usuarios(id_usuario) ON DELETE CASCADE,
  id_mascota INT REFERENCES mascotas(id_mascota) ON DELETE SET NULL,
  id_empleado INT REFERENCES empleados(id_empleado) ON DELETE SET NULL,
  id_servicio INT REFERENCES servicios(id_servicio) ON DELETE SET NULL,
  fecha_cita TIMESTAMPTZ NOT NULL,
  estado_cita estado_cita DEFAULT 'programada',
  motivo TEXT,
  observaciones TEXT,
  costo DECIMAL(10,2) DEFAULT 0.00,
  fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  creado_por INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL
);
COMMENT ON TABLE citas IS 'Sistema de citas; referencia a servicios por id_servicio';

-- ADOPCIONES
CREATE TABLE adopciones (
  id_adopcion SERIAL PRIMARY KEY,
  id_usuario INT NOT NULL REFERENCES usuarios(id_usuario) ON DELETE CASCADE,
  id_mascota INT REFERENCES mascotas(id_mascota) ON DELETE SET NULL,
  fecha_solicitud TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  fecha_entrega TIMESTAMPTZ,
  estado estado_adopcion DEFAULT 'en_proceso',
  estado_solicitud estado_solicitud_adopcion NOT NULL DEFAULT 'en_revision',
  documentos_verificados BOOLEAN DEFAULT FALSE,
  motivo_devolucion TEXT,
  fecha_devolucion TIMESTAMPTZ,
  observaciones TEXT,
  ubicacion_en_hogar TEXT,
  motivo_adopcion TEXT,
  procesado_por INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL,
  CONSTRAINT check_fecha_entrega_posterior
    CHECK (fecha_entrega IS NULL OR fecha_entrega >= fecha_solicitud)
);

-- una solicitud abierta por mascota
CREATE UNIQUE INDEX uq_adopcion_mascota_abierta
ON adopciones(id_mascota)
WHERE (estado_solicitud IN ('en_revision','aprobada') AND fecha_entrega IS NULL);
COMMENT ON TABLE adopciones IS 'Registro de procesos de adopción y su solicitud';

-- RESOLUCIONES DE ADOPCIÓN (aceptada/rechazada con motivo y fecha)
CREATE TABLE resoluciones_adopcion (
  id_resolucion SERIAL PRIMARY KEY,
  id_adopcion INT NOT NULL REFERENCES adopciones(id_adopcion) ON DELETE CASCADE,
  decision decision_adopcion NOT NULL,
  motivo TEXT,
  fecha_resolucion TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
  resuelto_por INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL,
  CONSTRAINT chk_motivo_si_rechazada
    CHECK ( (decision = 'rechazada' AND motivo IS NOT NULL) OR decision = 'aceptada' )
);
COMMENT ON TABLE resoluciones_adopcion IS 'Historial de decisiones de adopción con motivo en rechazos';

-- SEGUIMIENTOS
CREATE TABLE seguimientos_adopcion (
  id_seguimiento SERIAL PRIMARY KEY,
  id_adopcion INT NOT NULL REFERENCES adopciones(id_adopcion) ON DELETE CASCADE,
  fecha_seguimiento DATE NOT NULL,
  estado_mascota VARCHAR(50) NOT NULL,
  observaciones TEXT,
  requiere_atencion BOOLEAN DEFAULT FALSE,
  siguiente_seguimiento DATE,
  realizado_por INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL,
  fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT check_siguiente_seguimiento_posterior
    CHECK (siguiente_seguimiento IS NULL OR siguiente_seguimiento >= fecha_seguimiento)
);

-- FOTOS EN SEGUIMIENTO
CREATE TABLE seguimiento_fotos (
  id_foto SERIAL PRIMARY KEY,
  id_seguimiento INT NOT NULL REFERENCES seguimientos_adopcion(id_seguimiento) ON DELETE CASCADE,
  url TEXT NOT NULL,
  descripcion TEXT,
  fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- DOCS GENERALES DE PERSONA
CREATE TABLE documentos_persona (
  id_documento SERIAL PRIMARY KEY,
  id_usuario INT NOT NULL REFERENCES usuarios(id_usuario) ON DELETE CASCADE,
  tipo_documento VARCHAR(50) NOT NULL,
  numero_documento VARCHAR(50),
  archivo_url VARCHAR(500),
  verificado BOOLEAN DEFAULT FALSE,
  fecha_verificacion TIMESTAMPTZ,
  verificado_por INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL,
  observaciones TEXT,
  fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- DOCS POR ADOPCIÓN
CREATE TABLE documentos_adopcion (
  id_doc SERIAL PRIMARY KEY,
  id_adopcion INT NOT NULL REFERENCES adopciones(id_adopcion) ON DELETE CASCADE,
  tipo VARCHAR(30) NOT NULL CHECK (tipo IN ('INE','ACTA_NAC','COMP_DOM')),
  verificado BOOLEAN DEFAULT FALSE,
  verificado_por INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL,
  fecha_verificacion TIMESTAMPTZ,
  observaciones TEXT,
  fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- EXPEDIENTES VETERINARIOS
CREATE TABLE expedientes_veterinarios (
  id_expediente SERIAL PRIMARY KEY,
  id_mascota INT REFERENCES mascotas(id_mascota) ON DELETE SET NULL,
  observaciones TEXT,
  fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  creado_por INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL
);
COMMENT ON TABLE expedientes_veterinarios IS 'Historial clínico de cada mascota';

-- CONSULTAS VETERINARIAS
CREATE TABLE consultas_veterinarias (
  id_consulta SERIAL PRIMARY KEY,
  id_expediente INT NOT NULL REFERENCES expedientes_veterinarios(id_expediente) ON DELETE CASCADE,
  id_empleado INT REFERENCES empleados(id_empleado) ON DELETE SET NULL,
  fecha_consulta TIMESTAMPTZ NOT NULL,
  motivo TEXT,
  sintomas TEXT,
  diagnostico TEXT,
  tratamiento TEXT,
  recomendaciones TEXT,
  peso_kg NUMERIC(5,2),
  temperatura_c NUMERIC(4,1),
  costo NUMERIC(10,2) DEFAULT 0.00,
  proxima_cita TIMESTAMPTZ,
  fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  creado_por INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL
);
COMMENT ON TABLE consultas_veterinarias IS 'Consultas veterinarias asociadas a cada expediente';

-- VACUNAS (Opción A: sin catálogo)
CREATE TABLE mascota_vacunas (
  id_mascota_vacuna SERIAL PRIMARY KEY,
  id_mascota INT NOT NULL REFERENCES mascotas(id_mascota) ON DELETE CASCADE,
  vacuna_nombre TEXT NOT NULL,
  id_empleado INT REFERENCES empleados(id_empleado) ON DELETE SET NULL,
  fecha_aplicacion DATE NOT NULL,
  lote VARCHAR(50),
  proxima_dosis DATE,
  observaciones TEXT,
  fecha_creacion TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  creado_por INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL,
  CONSTRAINT uq_vacuna_evento UNIQUE (id_mascota, vacuna_nombre, fecha_aplicacion)
);
COMMENT ON TABLE mascota_vacunas IS 'Historial de vacunación con nombre libre';

-- NOTIFICACIONES (referencia a catálogo)
CREATE TABLE notificaciones (
  id_notificacion SERIAL PRIMARY KEY,
  id_usuario INT NOT NULL REFERENCES usuarios(id_usuario) ON DELETE CASCADE,
  tipo_notificacion tipo_notificacion NOT NULL,
  titulo TEXT NOT NULL,
  mensaje TEXT NOT NULL,
  creada_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE permisos (
    id_permiso SERIAL PRIMARY KEY,
    nombre_permiso VARCHAR(100) UNIQUE NOT NULL
);

CREATE TABLE roles_permisos (
    id_rol INT REFERENCES roles(id_rol),
    id_permiso INT REFERENCES permisos(id_permiso),
    PRIMARY KEY (id_rol, id_permiso)
);
